\section{Research Experience}
\begin{joblong}{\textsc{ImmersedPrivacy}: Evaluating VLMs' Privacy Awareness in the Physical World}{Jan 2026 - May 2026}{Georgia Institute of Technology \ $|$ \ First Author \ $|$ \ \textnormal{\textbf{NeurIPS 2026}} \ $|$ \ \href{https://immersed-privacy.github.io}{\textnormal{\faGlobe\ \texttt{Website}}}}{Dr. Pan Li}
\item Built a \textbf{reusable data generation and evaluation pipeline} in Unity for agent privacy, producing \textbf{400+ interactive audio-visual scenarios across 40+ physical environments} with configurable layouts and social contexts.
\item Established evaluation quality through \textbf{human-annotated baselines} and controlled scenario augmentation, separating perception failures from privacy reasoning for \textbf{attribution analysis}.
\item Benchmarked \textbf{12 VLMs} and exposed a \textbf{gap between privacy knowledge and behavior}: even the best model jointly satisfied task and privacy requirements in only \textbf{51\%} of conflicting-command cases.
\end{joblong}
\begin{joblong}{{Beyond Steering Vector: Flow-based Activation Steering}}{Mar 2026 - May 2026}{Georgia Institute of Technology \ $|$ \ First Author \ $|$ \ \textnormal{\textbf{NeurIPS 2026}} \ $|$ \href{https://flas-ai.github.io}{\textnormal{\faGlobe\ \texttt{Website}}}}{Dr. Chao Zhang}
\item Introduced \textsc{FLAS}, a \textbf{flow-based activation steering method} that learns concept-conditioned velocity fields to adapt interventions to each token's evolving activation state.
\item Revealed \textbf{curved, multi-step, token-dependent steering} trajectories through activation-space analysis, providing empirical evidence that challenges the \textbf{linear representation hypothesis} in activation steering.
\item Achieved \textbf{zero-shot generalization} to unseen concepts without per-concept tuning, becoming the first learned steering method to consistently outperform prompting on AxBench across Gemma-2-2B/9B.
\end{joblong}
\begin{joblong}{\textsc{SEAD}: A State-Based Perspective on Attack and Defense in Tool-Using Agents}{Jul 2026 - Sept 2026}{Georgia Institute of Technology \ $|$ \ First Author \ $|$ \ \href{https://everywheresafety.github.io/sead/}{\textnormal{\faGlobe\ \texttt{Website}}}}{Dr. Pan Li}
\item Formulated agent attack and defense as a \textbf{POMDP} and implemented an \textbf{agent safety testing sandbox} with \textbf{resettable environments, trajectory replay, and executable checks} across filesystem, terminal, database, and web tasks.
\item Developed \textsc{DART}, an \textbf{evolving attack agent} that refines strategies through execution feedback and tree search, using a \textbf{27B model} to attack commercial agents and achieving \textbf{73.3\% attack success} against GPT-5.6.
\item Developed \textsc{SAGE}, a \textbf{defender agent}, through \textbf{agentic post-training}, which actively \textbf{investigates environment} for safety evidence and reduces attack success to \textbf{at most 4.0\%} under various adaptive attacks.
\end{joblong}
\begin{joblongnoadvisor}{\textsc{AcFlow}: Controlling T2I DiTs via Learned Conditional Activation Flow}{Jul 2026 - Sept 2026}{Georgia Institute of Technology \ $|$ \ First Author \ $|$ \ \href{https://github.com/Nove1yst/AcFlow}{\textnormal{\faGlobe\ \texttt{Website}}}}
\item Developed \textsc{AcFlow} as a \textbf{scalable, text-conditioned, continuous-strength control module} for frozen image generators, enabling control toward a \textbf{general-purpose controller} for diffusion transformers.
\item Scaled concept coverage without per-concept fitting: one controller covers \textbf{15,000+ styles} or \textbf{1,000+ concepts} and \textbf{generalizes} to unseen concepts.
\item Validated the control interface on \textbf{FLUX and Z-Image}, achieving the \textbf{best intervention--preservation trade-off} among evaluated baselines.
\end{joblongnoadvisor}
