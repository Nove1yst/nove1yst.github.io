\section{Research Experience}
\begin{joblong}{\textsc{SEAD}: A State-Based Perspective on Attack and Defense in Tool-Using Agents}{Jul 2026 - Sept 2026}{Georgia Institute of Technology \ $|$ \ Co-first Author \ $|$ \ \href{https://everywheresafety.github.io/sead/}{\textnormal{\faGlobe\ \texttt{Website}}}}{Dr. Pan Li}
\item Formulated attack and defense for tool-using agents as \textbf{partially observed state control}, grounding safety in persistent environment changes and harmful-state reachability.
\item Developed \textsc{DART} and \textsc{SAGE} for adversarial evaluation and \textbf{state-aware pre-execution defense}, supported by replayable tool environments and task-specific executable checks.
\item \textsc{SAGE} preserved \textbf{95.79\%} of benign trajectories and intercepted \textbf{92.73\%} of harmful paths in recorded evaluations; reduced executable attack success from \textbf{48.0\% to 4.0\%} in online evaluation.
\end{joblong}
\begin{joblongnoadvisor}{\textsc{AcFlow}: Controlling T2I DiTs via Learned Conditional Activation Flow}{Jul 2026 - Sept 2026}{Georgia Institute of Technology \ $|$ \ First Author \ $|$ \ \href{https://github.com/Nove1yst/AcFlow}{\textnormal{\faGlobe\ \texttt{Website}}}}
\item Introduced \textsc{AcFlow}, a learned concept-conditioned \textbf{velocity field} that steers image-token activations in frozen text-to-image diffusion transformers through \textbf{token-specific, activation-dependent updates}.
\item Enabled \textbf{continuous} style control and concept suppression through adjustable flow horizons, with weights \textbf{shared across concepts} within each task family and \textbf{zero-shot generalization}.
\item Achieved the \textbf{best style--content trade-off} among evaluated baselines.
\end{joblongnoadvisor}
\begin{joblong}{\textsc{ImmersedPrivacy}: Evaluating VLMs'Privacy Awareness in the Physical World}{Jan 2026 - May 2026}{Georgia Institute of Technology \ $|$ \ First Author \ $|$ \ \href{https://immersed-privacy.github.io}{\textnormal{\faGlobe\ \texttt{Website}}}}{Dr. Pan Li}
\item Developed \textsc{ImmersedPrivacy}, a first-of-its-kind \textbf{interactive Visual+Audio benchmark} to evaluate VLMs' privacy awareness in dynamic physical environments.
\item Engineered a three-tier hierarchical framework built on a customized Unity simulator featuring 53 complex scenarios and 500+ procedurally generated instances.
\item Implemented an \textbf{interactive evaluation pipeline}, enabling agents to actively navigate environments and resolve task-privacy conflicts through multi-turn, closed-loop decision making.
\end{joblong}
\begin{joblong}{{Beyond Steering Vector: Flow-based Activation Steering}}{Mar 2026 - May 2026}{Georgia Institute of Technology \ $|$ \ Co-first Author \ $|$ \href{https://flas-ai.github.io}{\textnormal{\faGlobe\ \texttt{Website}}}}{Dr. Chao Zhang}
\item Introduced \textsc{FLAS}, a concept-conditioned velocity field that steers frozen LLM activations through adaptive, multi-step, and position-sensitive flows using positive examples alone.
\item Designed a lightweight FlowBlock architecture with cross-attention and Euler integration, enabling zero-shot control of unseen concepts and continuous steering-strength adjustment without per-concept tuning.
\item Achieved held-out HMean scores of \textbf{1.015/1.113} on Gemma-2-2B/9B-IT, making \textsc{FLAS} the first learned steering method to consistently outperform in-context prompting on AxBench.
\end{joblong}
